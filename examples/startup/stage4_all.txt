ls

ls projects

ls projects/emulator/src/main.py

# cd: относительный и абсолютный путь, '..', без аргумента

pwd

cd

# head: 10 строк по умолчанию, -n, -n больше длины файла, -n 0

head -n 100 home/user/todo.txt
head -n 0 numbers.txt

# uniq: обычный режим, -c, -d, -u и сочетание -c -d
uniq dups.txt
uniq -c dups.txt
uniq -d dups.txt
uniq -u dups.txt
uniq -c -d dups.txt
